> **不要用 `pgrep -f "docker compose build"` 判断构建是否结束。** sshd 为每条远端命令起的包裹 shell，其 cmdline 里就含有 `docker compose build` 这个子串，而 procps 的 pgrep **只排除自身 PID、不排除父 shell**，于是它永远匹配到自己、永远打印 `BUILDING`，轮询无限循环。已在 `ubuntu:22.04 + procps` 实测复现：`bash -c 'pgrep -af "docker compose build"'` 返回两行包裹 shell、`rc=0`；换成字符类 `[d]ocker compose build` 后返回 `rc=1`。（macOS 的 BSD pgrep 不复现，本机自测会误判成「没问题」。）下面改用**哨兵文件 + 退出码文件**，与 pgrep 的实现细节完全无关。

#: import of ``seed.preset_characters.SYSTEM_USER_ID``: this module is imported
#: by ``app/models/resident.py`` and must stay import-free, while ``seed``

#: fails the moment the two drift.
SYSTEM_CREATOR_ID = "00000000-0000-0000-0000-000000000001"

#: The literal ``creator_id`` written by the admin preset endpoint
#: (``app/routers/admin/residents.py:281``). Not a ``users`` row id.
ADMIN_CREATOR_ID = "system"

#: ``creator_id`` values that are *not* a real player account.
NON_USER_CREATOR_IDS = frozenset({SYSTEM_CREATOR_ID, ADMIN_CREATOR_ID})

#: ``meta_json["origin"]`` values written by the player-facing creation paths:
#: ``app/forge/pipeline.py:163`` (forge), ``app/forge/legacy_pipeline.py:141``
#: (forge) and ``:294`` (quick_forge), ``app/routers/residents.py:177,276``
#: (import).
UGC_ORIGINS = frozenset({"forge", "quick_forge", "import"})

#: ``meta_json["origin"]`` written by the admin preset path
#: (``app/routers/admin/residents.py:148``) — explicitly not UGC.
ADMIN_ORIGIN = "preset"


def ugc_provenance(meta_json, creator_id: str | None) -> str:
    """Classify one resident's *provenance* from its two durable columns.

    The single shared answer to "who is UGC", imported by both the one-off
    backfill (``scripts/civic_backfill.py``) and the promotion task. Writing it
    twice guarantees drift: the backfill demotes a batch, the promoter decides
    part of that batch is not UGC and therefore never promotes it — permanent
    second-class citizens, which is the very problem being fixed.

    Returns one of:

    ``"builtin"``
        Seed cast (``creator_id == SYSTEM_CREATOR_ID``).
    ``"admin"``
        Admin-created preset (``creator_id == "system"`` or
        ``origin == "preset"``).
    ``"ugc"``
        Player-authored: a known UGC ``origin`` **or** a ``creator_id`` that is
        a real user account id.
    ``"unknown"``
        Neither signal fired — ``origin`` missing/unrecognised *and*
        ``creator_id`` is NULL (migration 045 nulls it when the account is
        deleted). Never auto-classified as UGC; the backfill lists these for a
        manual roll-call instead (spec §6 硬约束 1 残差).

    Order matters: ``builtin``/``admin`` are decided before ``ugc`` so a seed
    resident carrying a stray ``origin`` can never be demoted. ``meta_json`` is
    a bare JSON column, so non-dict values are tolerated rather than raising.
    """
    meta = meta_json if isinstance(meta_json, dict) else {}
    origin = meta.get("origin")
    if creator_id == SYSTEM_CREATOR_ID:
        return "builtin"
    if creator_id == ADMIN_CREATOR_ID or origin == ADMIN_ORIGIN:
        return "admin"
    if origin in UGC_ORIGINS:
        return "ugc"
    if creator_id is not None and creator_id not in NON_USER_CREATOR_IDS:
        return "ugc"
    return "unknown"

#!/usr/bin/env python3
"""T2 存量回填:把 07-25 泄漏的 UGC 居民从 ``npc`` 降回 ``resident``。

2026-07-25,玩家创作(forge / import / quick_forge)的居民依赖 ``Resident``

事故根因正是「手工脚本自带 id 列表绕过 ``find_targets``」,12 个玩家角色被连带
删除。所以这里的规矩是:目标集由代码算、防呆先于第一条 UPDATE、拒绝时是真正的
no-op、默认 dry-run。

用法(vm212 的 api 容器内跑,DATABASE_URL 由 compose 注入)::

    # 1) 先看名单,什么都不写(默认行为)
    docker compose exec -T api python scripts/civic_backfill.py
    # 2) 名单核对无误后再写库
    docker compose exec -T api python scripts/civic_backfill.py --apply

三道防呆,任何一道不过都在第一条 UPDATE 之前 raise ``CivicBackfillRefused``:

1. **不可重放** —— 成功后往 ``system_config`` 写 ``civic_backfill_done``;标记
   已存在且没带 ``--force-rerun`` 就拒绝退出。F2 的公民权晋升上线后重放一次 =
   把刚晋升的公民整批剥权,而且零告警。
2. **无 open poll** —— 降权会缩小法定人数分母。票开着的时候改分母,等于让已经
   投出的票按新分母重新判定门槛。``--allow-open-poll`` 可显式绕过。
3. **不碰玩家化身** —— 查 ``users.player_resident_id`` 复核,而不是信
   ``resident_type``(对标 ``seed/reset_builtin_residents.PlayerPurgeRefused``:
   raise 而非静默跳过,查库而非信传入对象)。

目标谓词见 ``app.services.civic_membership.ugc_provenance``——T2 与 F2 共用同
一份,两边各写一份必然漂移。``unknown``(origin 缺失且 creator_id 为 NULL,迁移
045 注销账号的残差)**不自动降权**,只列出来交人工点名。

标记里带世界时锚点(``completed_at_world``)。spec §7 的降级路径靠它:F2 首次
读取时,无历史行的 UGC 的 anchor 取
``max(real_to_world(created_at), completed_at_world)``——否则一个已在镇 200
世界日的 UGC 被降权后,F2 开闸当晚条件①立刻重新满足,降权整批走过场。
"""
from __future__ import annotations

import argparse
import asyncio
import json
import os
import sys

# `python scripts/civic_backfill.py` 直接跑时保证 `app` 可导入
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from sqlalchemy import inspect as sa_inspect, select, text  # noqa: E402

from app import world_clock  # noqa: E402
from app.models.resident import Resident  # noqa: E402
from app.models.season import Poll  # noqa: E402
from app.models.user import User  # noqa: E402
from app.services.civic_membership import (  # noqa: E402
    UGC_RESIDENT_TYPE,
    ugc_provenance,
)
from app.services.config_service import ConfigService  # noqa: E402

#: 不可重放标记。写进 system_config(value 是 String(2000),所以标记里不放
#: slug 名单——审计名单走 stdout 与 Task 12 的报告)。
BACKFILL_MARKER_KEY = "civic_backfill_done"
BACKFILL_MARKER_GROUP = "civic"
BACKFILL_MARKER_VERSION = 1

#: F2 交付的档位历史表(spec §7 ①)。本脚本运行时它可能还不存在。
HISTORY_TABLE = "civic_standing_history"
#: 与 F2 的列名契约。写成候选名单而不是单一名字,是因为 F2 的建表迁移与本脚本
#: 并行开发;认不出来就 raise(见 `_history_columns`),绝不猜。
_HISTORY_SUBJECT_COLUMNS = ("resident_slug", "slug", "resident_id")
#: 刻意**不含**裸 ``"to"``:它是 SQL 保留字,拼进 ``WHERE to = :tier`` 会得到一个
#: 裸 ``ProgrammingError``(退出码 1)而不是 ``CivicBackfillRefused``(退出码 2),
#: 而运维脚本对 1 和 2 的语义解读完全不同。实测 sqlite:
#: ``SELECT … WHERE to = 'citizen'`` → ``near "to": syntax error``;PostgreSQL 同理。
#: F2 若真用了 ``to`` 这个列名,本脚本会走 fail-closed 的 raise 分支——那正是想要的。
_HISTORY_TIER_COLUMNS = ("to_standing", "to_tier")
CITIZEN_TIER = "citizen"


class CivicBackfillRefused(RuntimeError):
    """一次回填被防呆拦下。

    永远 raise 在第一条 UPDATE 之前,所以拒绝是真正的 no-op——包括同一次调用
    里那些本来合法的目标。对标 ``PlayerPurgeRefused``。
    """


# ---------------------------------------------------------------------------
# 已晋升者(F2 的 to=citizen 历史行)
# ---------------------------------------------------------------------------

async def _table_exists(db, table_name: str) -> bool:
    def _check(sync_conn) -> bool:
        return sa_inspect(sync_conn).has_table(table_name)

    conn = await db.connection()
    return await conn.run_sync(_check)


async def _history_columns(db) -> tuple[str, str]:
    """解析 ``civic_standing_history`` 的 (主体列, 档位列)。

    认不出来就 raise。宁可整次回填停下,也不能把「已晋升者」当成没晋升过——那
    等于在 F2 上线后把刚拿到公民权的人再剥一次,而且零告警。
    """
    def _cols(sync_conn) -> set[str]:
        return {c["name"] for c in sa_inspect(sync_conn).get_columns(HISTORY_TABLE)}

    conn = await db.connection()
    names = await conn.run_sync(_cols)
    subject = next((c for c in _HISTORY_SUBJECT_COLUMNS if c in names), None)
    tier = next((c for c in _HISTORY_TIER_COLUMNS if c in names), None)
    if subject is None or tier is None:
        raise CivicBackfillRefused(
            f"{HISTORY_TABLE} 存在但列名不认识(实际列 {sorted(names)}):"
            f"找不到主体列 {list(_HISTORY_SUBJECT_COLUMNS)} 或档位列 "
            f"{list(_HISTORY_TIER_COLUMNS)}。F2 若改了表结构,请同步改本脚本的"
            "候选名单——宁可停下,也不能把已晋升者当成没晋升过而降权。"
        )
    return subject, tier


async def promoted_ids(db, residents: list[Resident]) -> set[str]:
    """已被 F2 晋升过(有 ``to=citizen`` 历史行)的居民 id 集合。

    表不存在 → 空集:没有表就没有任何晋升发生过(T2 先于 F2 合入的窗口)。
    """
    if not residents or not await _table_exists(db, HISTORY_TABLE):
        return set()
    subject_col, tier_col = await _history_columns(db)
    # 标识符来自上面的固定白名单,不是用户输入;取值仍走绑定参数。仍然逐个加双
    # 引号:白名单里将来若混进一个 SQL 保留字,裸拼会炸成 ProgrammingError(退出
    # 码 1),而本脚本对外承诺的失败形态是 CivicBackfillRefused(退出码 2)。
    rows = (await db.execute(
        text(f'SELECT DISTINCT "{subject_col}" AS subject FROM "{HISTORY_TABLE}" '
             f'WHERE "{tier_col}" = :tier'),
        {"tier": CITIZEN_TIER},
    )).all()
    subjects = {r[0] for r in rows if r[0] is not None}
    return {r.id for r in residents if r.id in subjects or r.slug in subjects}


# ---------------------------------------------------------------------------
# 目标集
# ---------------------------------------------------------------------------

async def find_targets(db) -> dict[str, list[Resident]]:
    """把当前 ``resident_type == "npc"`` 的居民按出身分桶。

    只扫 ``"npc"``:泄漏正是「UGC 被写成 npc」,已经是 ``"resident"`` 的人无需
    再动,``"player"`` / ``"preset"`` 不在政治层里。返回的键固定为
    ``ugc`` / ``builtin`` / ``admin`` / ``unknown`` / ``promoted``,只有 ``ugc``
    是回填目标。
    """
    residents = (await db.execute(
        select(Resident).where(Resident.resident_type == "npc").order_by(Resident.slug)
    )).scalars().all()
    promoted = await promoted_ids(db, residents)
    buckets: dict[str, list[Resident]] = {
        "ugc": [], "builtin": [], "admin": [], "unknown": [], "promoted": [],
    }
    for r in residents:
        if r.id in promoted:
            buckets["promoted"].append(r)
            continue
        buckets[ugc_provenance(r.meta_json, r.creator_id)].append(r)
    return buckets


# ---------------------------------------------------------------------------
# 三道防呆(全部在第一条 UPDATE 之前)
# ---------------------------------------------------------------------------

async def _assert_not_replayed(db, *, force_rerun: bool) -> dict | None:
    marker = await ConfigService(db).get(BACKFILL_MARKER_KEY)
    if marker is not None and not force_rerun:
        raise CivicBackfillRefused(
            f"{BACKFILL_MARKER_KEY} 已存在({marker}):本回填是一次性数据变更。"
            "F2 的公民权晋升上线后重放一次 = 把刚晋升的公民整批剥权且零告警。"
            "确需重跑请先看 dry-run 名单,再显式加 --force-rerun。"
        )
    return marker


async def _assert_no_open_poll(db, *, allow_open_poll: bool) -> list[str]:
    open_ids = [str(i) for i in (await db.execute(
        select(Poll.id).where(Poll.status == "open").order_by(Poll.id)
    )).scalars().all()]
    if open_ids and not allow_open_poll:
        raise CivicBackfillRefused(
            f"仍有 {len(open_ids)} 张 open poll(id={open_ids}):降权会缩小法定"
            "人数分母,在票开着的时候改分母 = 已投出的票按新分母重新判定门槛。"
            "等这些 poll 关掉再跑,或显式加 --allow-open-poll。"
        )
    return open_ids


async def _assert_no_players(db, ids: list[str]) -> None:
    """拒绝含玩家化身的目标名单。

    两条都查库,不信传入对象:``resident_type == "player"`` 是显式那一半,
    ``users.player_resident_id`` 是「化身被误写成 npc」那一半——后者正是本脚本
    真正的风险面,因为 ``find_targets`` 只扫 npc。
    """
    if not ids:
        return
    typed = (await db.execute(
        select(Resident.slug).where(
            Resident.id.in_(ids), Resident.resident_type == "player")
    )).scalars().all()
    linked = (await db.execute(
        select(Resident.slug)
        .join(User, User.player_resident_id == Resident.id)
        .where(Resident.id.in_(ids))
    )).scalars().all()
    offenders = sorted(set(typed) | set(linked))
    if offenders:
        raise CivicBackfillRefused(
            f"拒绝降权 {len(offenders)} 个玩家化身:{', '.join(offenders)}。"
            "玩家化身由 users.player_resident_id 单值 FK 管辖,不属于任何 UGC "
            "回填范围;它出现在 npc 目标集里本身就是一处数据损坏,先查清来源。"
        )


# ---------------------------------------------------------------------------
# 回填
# ---------------------------------------------------------------------------

async def backfill(
    db, *, apply: bool = False, force_rerun: bool = False,
    allow_open_poll: bool = False,
) -> dict:
    """扫描并(可选地)把泄漏的 UGC 居民降回 ``resident``。

    返回的报告 dict::

        {"apply": bool,
         "prior_marker": dict | None,
         "open_poll_ids": list[str],
         "targets": [{"slug", "name", "origin", "creator_id"}, ...],
         "skipped": {"builtin"|"admin"|"unknown"|"promoted": [slug, ...]},
         "marker": dict | None}

    ``apply=False``(默认)一个字节都不写。``apply=True`` 时降权与标记在
    **同一次 commit** 里落盘:``ConfigService.set`` 内部 commit,而降权是同一个
    session 的待写变更,由那次 commit 一并提交——不会出现「降了权但没留标记」
    的可重放状态。
    """
    prior_marker = await _assert_not_replayed(db, force_rerun=force_rerun)
    open_polls = await _assert_no_open_poll(db, allow_open_poll=allow_open_poll)
    buckets = await find_targets(db)
    targets = buckets["ugc"]
    await _assert_no_players(db, [r.id for r in targets])

    result: dict = {
        "apply": apply,
        "prior_marker": prior_marker,
        "open_poll_ids": open_polls,
        "targets": [{
            "slug": r.slug,
            "name": r.name,
            "origin": (r.meta_json or {}).get("origin")
                      if isinstance(r.meta_json, dict) else None,
            "creator_id": r.creator_id,
        } for r in targets],
        "skipped": {k: [r.slug for r in v]
                    for k, v in buckets.items() if k != "ugc"},
        "marker": None,
    }
    if not apply:
        return result

    for r in targets:
        r.resident_type = UGC_RESIDENT_TYPE
    # 一次 now_world(),三个字段从同一个瞬间派生。一个世界日只有 6 个真实小时,
    # 两次独立取时可能跨日界,标记会自相矛盾。
    world_now = world_clock.now_world()
    marker = {
        "version": BACKFILL_MARKER_VERSION,
        "completed_at_real": world_clock.now_real().isoformat(),
        "completed_at_world": world_now.isoformat(),
        "world_date_key": world_now.strftime("%Y-%m-%d"),
        "demoted_count": len(targets),
    }
    await ConfigService(db).set(
        BACKFILL_MARKER_KEY, marker,
        group=BACKFILL_MARKER_GROUP, updated_by="civic_backfill",
    )
    result["marker"] = marker
    return result
