```python
def get_location_id_by_name(name: str | None) -> str | None:
    """Reverse-lookup a location id by its display name (first match)."""
    if not name:
        return None
    for loc_id, loc in LOCATIONS.items():
        if loc.get("name") == name:
            return loc_id
    return None
```

`plan_target.py`(Create):
```python
"""Server-side plan/decision target resolution (realism P0-1).

Model-reported coordinates are ignored by design (hallucinated tiles are half
the "plan doesn't move" bug). The target tile is always derived from a known
location: prefer an explicit location id (slug), else the display name.
"""
from __future__ import annotations

from app.agent.map_data import (
    get_location_by_id,
    get_location_id_by_name,
    get_valid_target_tile,
)


def resolve_target_tile(
    target_slug: str | None,
    location_name: str | None,
) -> tuple[int, int] | None:
    if target_slug and get_location_by_id(target_slug) is not None:
        return get_valid_target_tile(target_slug)
    loc_id = get_location_id_by_name(location_name)
    if loc_id is not None:
        return get_valid_target_tile(loc_id)
    return None
```

```python
_MOVEMENT_ACTIONS = {ActionType.WANDER, ActionType.VISIT_DISTRICT, ActionType.GO_HOME}

def _movement_memory(ctx) -> tuple[str, dict]:
    from app.agent.plan_target import resolve_target_tile
    ar = ctx.action_result
    res = ctx.resident
    intent = ar.action.value
    target_slug = ar.target_slug
    target_tile = ar.target_tile or resolve_target_tile(target_slug, None)
    loc = get_location_at(res.tile_x, res.tile_y)
    loc_name = loc["name"] if loc else "户外"
    moved = ctx.new_tile is not None and ctx.new_tile != _prev_tile(ctx)
    arrived = target_tile is not None and (res.tile_x, res.tile_y) == tuple(target_tile)
    if ar.action == ActionType.GO_HOME:
        dest = "家"
    else:
        dest = loc_name if arrived else (_target_name(target_slug) or "某处")
    if arrived:
        text = f"到达了{dest}"
    elif moved:
        text = f"正在前往{dest}"
    else:
        # 未发生位移，不写"去了"——记录停留
        text = f"在{loc_name}停留"
    return text, {"intent": intent, "target": target_slug, "moved": moved, "arrived": arrived}
```

```python
    async def _search_events_scored(self, resident_id, query_embedding, limit=10):
        from app.config import settings
        candidates = await self._fetch_event_candidates(resident_id, cap=max(limit*3, 30))
        now = datetime.now(UTC)
        wr = settings.realism_retrieval_relevance_weight
        wc = settings.realism_retrieval_recency_weight
        wi = settings.realism_retrieval_importance_weight
        tau_base = settings.realism_recency_tau_hours
        scored = []
        for m in candidates:
            rel = _cosine(query_embedding, m.embedding) if m.embedding else 0.0
            imp = m.importance or 0.0
            created = m.created_at if m.created_at.tzinfo else m.created_at.replace(tzinfo=UTC)
            dt_h = max((now - created).total_seconds() / 3600.0, 0.0)
            tau = tau_base * (1 + imp)
            recency = math.exp(-dt_h / tau) if tau > 0 else 0.0
            scored.append((wr*rel + wc*recency + wi*imp, m))
        scored.sort(key=lambda t: t[0], reverse=True)
        return [m for _, m in scored[:limit]]
```
